import pygame import random import sys import math # إعداد الأساسيات pygame.init() WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Space Invaders: Elite Generation") clock = pygame.time.Clock() # الألوان COLOR_BG = (10, 10, 25) COLOR_PLAYER = (0, 255, 204) COLOR_ENEMY_BASIC = (255, 51, 51) COLOR_ENEMY_FAST = (255, 153, 51) COLOR_LASER_PLAYER = (0, 255, 102) COLOR_LASER_ENEMY = (255, 51, 153) COLOR_TEXT = (240, 240, 240) COLOR_SHIELD = (0, 153, 255) # الخطوط font_sub = pygame.font.SysFont("Courier New", 20, bold=True) font_main = pygame.font.SysFont("Courier New", 36, bold=True) font_title = pygame.font.SysFont("Courier New", 60, bold=True) class Player: def __init__(self): self.width = 50 self.height = 40 self.x = WIDTH // 2 - self.width // 2 self.y = HEIGHT - 70 self.speed = 6 self.max_health = 100 self.health = 100 self.cooldown = 0 self.cooldown_max = 15 # سرعة الإطلاق البدائية self.score = 0 self.double_shot = False self.powerup_timer = 0 def move(self, keys): if (keys[pygame.K_LEFT] or keys[pygame.K_a]) and self.x > 0: self.x -= self.speed if (keys[pygame.K_RIGHT] or keys[pygame.K_d]) and self.x < WIDTH - self.width: self.x += self.speed def fire(self, keys, player_lasers): if self.cooldown > 0: self.cooldown -= 1 if (keys[pygame.K_SPACE] or keys[pygame.K_UP]) and self.cooldown == 0: if self.double_shot: player_lasers.append(Laser(self.x + 5, self.y, -8, COLOR_LASER_PLAYER)) player_lasers.append(Laser(self.x + self.width - 10, self.y, -8, COLOR_LASER_PLAYER)) else: player_lasers.append(Laser(self.x + self.width // 2 - 2, self.y, -8, COLOR_LASER_PLAYER)) self.cooldown = self.cooldown_max def draw(self, surface): # رسم مركبة فضائية متطورة بالخطوط والأشكال points = [ (self.x + self.width // 2, self.y), (self.x, self.y + self.height), (self.x + self.width // 4, self.y + self.height * 0.75), (self.x + self.width * 0.75, self.y + self.height * 0.75), (self.x + self.width, self.y + self.height) ] pygame.draw.polygon(surface, COLOR_PLAYER, points) if self.double_shot: pygame.draw.circle(surface, COLOR_SHIELD, (self.x + self.width // 2, self.y + self.height // 2), 30, 2) # شريط الصحة أسفل المركبة health_bar_width = self.width health_ratio = self.health / self.max_health pygame.draw.rect(surface, (100, 0, 0), (self.x, self.y + self.height + 10, health_bar_width, 5)) pygame.draw.rect(surface, (0, 255, 0), (self.x, self.y + self.height + 10, health_bar_width * health_ratio, 5)) class Enemy: def __init__(self, x, y, enemy_type): self.x = x self.y = y self.type = enemy_type if self.type == "fast": self.width, self.height = 30, 25 self.color = COLOR_ENEMY_FAST self.speed_x = random.choice([-4, -3, 3, 4]) self.speed_y = 0.5 self.health = 1 self.points = 25 else: self.width, self.height = 40, 30 self.color = COLOR_ENEMY_BASIC self.speed_x = random.choice([-2, -1, 1, 2]) self.speed_y = 0.3 self.health = 2 self.points = 10 def update(self): self.x += self.speed_x self.y += self.speed_y # الارتداد عن الجوانب if self.x <= 0 or self.x >= WIDTH - self.width: self.speed_x *= -1 def draw(self, surface): # شكل العدو الفضائي rect = pygame.Rect(self.x, self.y, self.width, self.height) pygame.draw.rect(surface, self.color, rect, border_radius=5) # رسم أعين الكائن الفضائي pygame.draw.circle(surface, COLOR_BG, (self.x + self.width // 4, self.y + self.height // 3), 4) pygame.draw.circle(surface, COLOR_BG, (self.x + self.width * 0.75, self.y + self.height // 3), 4) class Laser: def __init__(self, x, y, speed_y, color): self.x = x self.y = y self.width = 4 self.height = 15 self.speed_y = speed_y self.color = color def update(self): self.y += self.speed_y def draw(self, surface): pygame.draw.rect(surface, self.color, (self.x, self.y, self.width, self.height)) class PowerUp: def __init__(self, x, y, pu_type): self.x = x self.y = y self.type = pu_type # "heal" أو "double" self.size = 15 self.color = (0, 255, 0) if pu_type == "heal" else COLOR_SHIELD def update(self): self.y += 3 def draw(self, surface): pygame.draw.circle(surface, self.color, (self.x, self.y), self.size) class Particle: """تأثيرات الانفجارات البصرية لقوة الـ Repo البصرية""" def __init__(self, x, y, color): self.x = x self.y = y self.color = color self.vx = random.uniform(-3, 3) self.vy = random.uniform(-3, 3) self.lifetime = random.randint(10, 25) def update(self): self.x += self.vx self.y += self.vy self.lifetime -= 1 def draw(self, surface): if self.lifetime > 0: pygame.draw.circle(surface, self.color, (int(self.x), int(self.y)), 3) # مصفوفة النجوم في الخلفية (Parallax Starfield) stars = [[random.randint(0, WIDTH), random.randint(0, HEIGHT), random.uniform(1, 3)] for _ in range(50)] def main(): player = Player() enemies = [] player_lasers = [] enemy_lasers = [] powerups = [] particles = [] wave = 0 game_state = "MENU" # MENU, PLAYING, GAME_OVER while True: keys = pygame.key.get_pressed() # إدارة الأحداث for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() if event.type == pygame.KEYDOWN: if game_state == "MENU" and event.key == pygame.K_SPACE: game_state = "PLAYING" player.reset_game() if hasattr(player, 'reset_game') else None elif game_state == "GAME_OVER" and event.key == pygame.K_SPACE: player = Player() enemies, player_lasers, enemy_lasers, powerups, particles = [], [], [], [], [] wave = 0 game_state = "PLAYING" # --- حالة القائمة الرئيسية (MENU) --- if game_state == "MENU": screen.fill(COLOR_BG) title = font_title.render("SPACE INVADERS", True, COLOR_PLAYER) sub = font_main.render("Press [SPACE] to Start", True, COLOR_TEXT) github_tag = font_sub.render("GitHub Premium Edition", True, (100, 100, 150)) screen.blit(title, (WIDTH//2 - title.get_width()//2, HEIGHT//3)) screen.blit(sub, (WIDTH//2 - sub.get_width()//2, HEIGHT//2)) screen.blit(github_tag, (WIDTH//2 - github_tag.get_width()//2, HEIGHT - 50)) pygame.display.flip() clock.tick(60) continue # --- حالة الخسارة (GAME OVER) --- if game_state == "GAME_OVER": screen.fill(COLOR_BG) title = font_title.render("GAME OVER", True, COLOR_ENEMY_BASIC) sub = font_main.render(f"Final Score: {player.score}", True, COLOR_TEXT) restart = font_sub.render("Press [SPACE] to Try Again", True, COLOR_PLAYER) screen.blit(title, (WIDTH//2 - title.get_width()//2, HEIGHT//3)) screen.blit(sub, (WIDTH//2 - sub.get_width()//2, HEIGHT//2)) screen.blit(restart, (WIDTH//2 - restart.get_width()//2, HEIGHT * 0.65)) pygame.display.flip() clock.tick(60) continue # --- حالة اللعب الفعلي (PLAYING) --- # 1. تحديث النجوم بالخلفية for star in stars: star[1] += star[2] if star[1] > HEIGHT: star[1] = 0 star[0] = random.randint(0, WIDTH) # 2. حركة اللاعب والإطلاق player.move(keys) player.fire(keys, player_lasers) if player.double_shot: player.powerup_timer -= 1 if player.powerup_timer <= 0: player.double_shot = False # 3. توليد موجات الأعداء (Wave System) if len(enemies) == 0: wave += 1 for row in range(3): for col in range(8): e_type = "fast" if row == 0 and wave > 1 else "basic" enemies.append(Enemy(col * 80 + 100, row * 50 + 60, e_type)) # 4. تحديث الأعداء وإطلاق ليزر الأعداء عشوائياً for enemy in enemies[:]: enemy.update() # فرصة عشوائية لكل عدو ليطلق النار if random.randint(1, 1500) == 7: enemy_lasers.append(Laser(enemy.x + enemy.width//2, enemy.y + enemy.height, 4, COLOR_LASER_ENEMY)) # إذا وصل العدو لأسفل الشاشة if enemy.y + enemy.height >= player.y: game_state = "GAME_OVER" # 5. تحديث المقذوفات والمؤثرات for laser in player_lasers[:]: laser.update() if laser.y < 0: player_lasers.remove(laser) for laser in enemy_lasers[:]: laser.update() if laser.y > HEIGHT: enemy_lasers.remove(laser) for pu in powerups[:]: pu.update() if pu.y > HEIGHT: powerups.remove(pu) for p in particles[:]: p.update() if p.lifetime <= 0: particles.remove(p) # 6. معالجة الاصطدامات (Collisions) # ليزر اللاعب يصيب الأعداء for laser in player_lasers[:]: for enemy in enemies[:]: if (laser.x > enemy.x and laser.x < enemy.x + enemy.width and laser.y > enemy.y and laser.y < enemy.y + enemy.height): if laser in player_lasers: player_lasers.remove(laser) enemy.health -= 1 # توليد جزيئات انفجار for _ in range(10): particles.append(Particle(enemy.x + enemy.width//2, enemy.y + enemy.height//2, enemy.color)) if enemy.health <= 0: player.score += enemy.points # فرصة إسقاط هدايا الدعم (Power-ups) if random.random() < 0.2: powerups.append(PowerUp(enemy.x + enemy.width//2, enemy.y, random.choice(["heal", "double"]))) enemies.remove(enemy) # ليزر الأعداء يصيب اللاعب for laser in enemy_lasers[:]: if (laser.x > player.x and laser.x < player.x + player.width and laser.y > player.y and laser.y < player.y + player.height): enemy_lasers.remove(laser) player.health -= 15 for _ in range(5): particles.append(Particle(laser.x, laser.y, COLOR_ENEMY_BASIC)) if player.health <= 0: game_state = "GAME_OVER" # اللاعب يلتقط هدايا الدعم for pu in powerups[:]: if (pu.x > player.x and pu.x < player.x + player.width and pu.y > player.y and pu.y < player.y + player.height): if pu.type == "heal": player.health = min(player.max_health, player.health + 30) elif pu.type == "double": player.double_shot = True player.powerup_timer = 400 # مدة تفعيل السلاح المزدوج powerups.remove(pu) # --- الرسم على الشاشة --- screen.fill(COLOR_BG) # رسم خلفية النجوم متحركة for star in stars: pygame.draw.circle(screen, (200, 200, 250), (int(star[0]), int(star[1])), 1) player.draw(screen) for enemy in enemies: enemy.draw(screen) for laser in player_lasers: laser.draw(screen) for laser in enemy_lasers: laser.draw(screen) for pu in powerups: pu.draw(screen) for p in particles: p.draw(screen) # لوحة العرض العلوية (HUD) ui_score = font_sub.render(f"SCORE: {player.score}", True, COLOR_TEXT) ui_wave = font_sub.render(f"WAVE: {wave}", True, COLOR_PLAYER) screen.blit(ui_score, (20, 20)) screen.blit(ui_wave, (WIDTH - ui_wave.get_width() - 20, 20)) pygame.display.flip() clock.tick(60) if __name__ == "__main__": main()
